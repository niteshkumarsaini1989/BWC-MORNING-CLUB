  <script>
    var currentUser = JSON.parse(sessionStorage.getItem('bwc_session_user')) || null;
    var coaches = [];

    function quickLogin(role, coachId, coachName) {
      try {
        if (role === 'ADMIN') {
          currentUser = { id: 'ADMIN', name: 'Master Admin', role: 'ADMIN' };
        } else {
          let matchedCoach = coaches.find(co => co.id === coachId || co.phone === coachId || (co.code && co.code.toLowerCase() === (coachId || '').toLowerCase()));
          if (!matchedCoach && coachId && coachId !== 'COACH_DEMO') {
            matchedCoach = coaches.find(co => co.id === coachId);
          }
          if (!matchedCoach) {
            matchedCoach = coaches.length > 0 && coachId !== 'COACH_DEMO' ? coaches[0] : { id: 'COACH_DEMO', name: coachName || 'Demo Coach', role: 'COACH' };
          }
          currentUser = { id: matchedCoach.id, name: matchedCoach.name, role: 'COACH' };
        }
        sessionStorage.setItem('bwc_session_user', JSON.stringify(currentUser));
        const authEl = document.getElementById('authOverlay');
        if (authEl) authEl.style.display = 'none';
        if (typeof applyRoleView === 'function') applyRoleView();
        if (typeof refreshUI === 'function') refreshUI();
        if (typeof checkDeviceView === 'function') checkDeviceView();
      } catch (err) {
        console.error('quickLogin error:', err);
        const authEl = document.getElementById('authOverlay');
        if (authEl) authEl.style.display = 'none';
      }
    }

    function quickLoginCoach() {
      const select = document.getElementById('loginCoachSelect');
      const coachId = select ? select.value : 'COACH_DEMO';
      let coachName = 'Demo Coach';
      if (select && select.selectedIndex >= 0) {
        coachName = select.options[select.selectedIndex].text.replace(/^[^\w\u0900-\u097F]+/, '').trim();
      }
      quickLogin('COACH', coachId, coachName);
    }
  </script>

    function renderDailySheet() {
      const selectedDate = document.getElementById('dailyAttDate').value;

      const mobileContainer = document.getElementById('attMobileCards');

                ${isPresent 
                  ? `<button class="btn btn-sm btn-success fw-bold px-3" onclick="toggleDailyAttendance('${selectedDate}', ${c.id}, false)">✔ Present</button>`
                  : `<button class="btn btn-sm btn-outline-danger fw-bold px-3" onclick="toggleDailyAttendance('${selectedDate}', ${c.id}, true)">Absent</button>`

              </div>
            </div>
          </div>
        `;

        tbody.innerHTML += `
          <tr class="${isPresent ? 'bg-success bg-opacity-10' : 'bg-danger bg-opacity-10'}">
            <td>${index + 1}</td>
            <td><span class="badge bg-secondary">${c.code}</span></td>
            <td class="text-start fw-bold">${c.name}</td>
            <td>${c.phone}</td>
            <td>${isPresent ? '<span class="badge bg-success">PRESENT</span>' : '<span class="badge bg-danger">ABSENT</span>'}</td>
            <td>
              ${isPresent
                ? `<button class="btn btn-sm btn-outline-danger" onclick="toggleDailyAttendance('${selectedDate}', ${c.id}, false)">Absent</button>`
                : `<button class="btn btn-sm btn-success px-3 fw-bold" onclick="toggleDailyAttendance('${selectedDate}', ${c.id}, true)">✔ Present</button>`
              }
            </td>
            <td><button class="btn btn-sm btn-outline-info py-0 px-2" onclick="generateAttendanceReportOnly(${c.id})"><i class="fa-solid fa-print"></i></button></td>

      document.getElementById('statPresentToday').innerText = presentCount;

    function renderMonthlySheet() {
      const monthStr = document.getElementById('monthlyAttPicker').value;
      const headerRow = document.getElementById('monthAttHeaderRow');
      const tbody = document.getElementById('monthAttTableBody');
      const list = getFilteredConsumers();

      const year = parseInt(monthStr.split('-')[0]);
      const monthIndex = parseInt(monthStr.split('-')[1]);
      const totalDays = new Date(year, monthIndex, 0).getDate();

      headerRow.innerHTML = '<th>Sr.</th><th class="text-start" style="min-width:140px;">Name</th>';
      for (let d = 1; d <= totalDays; d++) headerRow.innerHTML += `<th style="width:30px; font-size:0.75rem;">${d}</th>`;
      headerRow.innerHTML += '<th>Total</th>';

      tbody.innerHTML = '';
      if (list.length === 0) {
        tbody.innerHTML = `<tr><td colspan="${totalDays + 3}" class="text-muted p-3">Koi consumer nahi mila.</td></tr>`;
        return;
      }

      list.forEach((c, index) => {
        let totalPresent = 0;
        let daysHTML = '';

        for (let d = 1; d <= totalDays; d++) {
          const attKey = `${monthStr}_${c.id}_${d}`;
          const isPresent = attendanceData[attKey] === true;
          if (isPresent) totalPresent++;

          daysHTML += `
            <td style="cursor:pointer;" class="${isPresent ? 'bg-success text-white fw-bold' : 'text-muted'}" 
                onclick="toggleMonthlyCell('${attKey}', ${!isPresent})">
              ${isPresent ? '✔' : '·'}
            </td>
          `;
        }

        tbody.innerHTML += `
          <tr>
            <td>${index + 1}</td>
            <td class="text-start fw-bold text-dark">${c.name}</td>
            ${daysHTML}
            <td><span class="badge bg-success">${totalPresent}</span></td>
          </tr>
        `;
      });
    }

    function toggleMonthlyCell(attKey, status) { db.ref('attendance/' + attKey).set(status); }

    function findMatchingProduct(keyword) {
      if (!keyword) return null;
      keyword = keyword.trim().toLowerCase();
      return masterProductsDB.find(p => 
        p.keys.toLowerCase().includes(keyword) ||
        p.name.toLowerCase().includes(keyword) ||
        p.id.toLowerCase() === keyword
      );
    }

    function handleProductTabKey(e) {
      if (e.key === 'Tab') {
        const input = e.target;
        const typedVal = input.value.trim();
        if (typedVal.length > 0) {
          const matched = findMatchingProduct(typedVal);
          if (matched) {
            input.value = matched.name;
            const parentRow = input.closest('.product-row-item');
            const hiddenId = parentRow.querySelector('.salesSelectedProductId');
            if (hiddenId) hiddenId.value = matched.id;
          }
        }
      }
    }

    function onProductSearchSelect(inputElement) {
      const searchVal = inputElement.value.trim().toLowerCase();
      const matched = findMatchingProduct(searchVal);
      const parentRow = inputElement.closest('.product-row-item');
      const hiddenId = parentRow.querySelector('.salesSelectedProductId');

      if (matched && hiddenId) {
        hiddenId.value = matched.id;
      } else if (hiddenId) {
        hiddenId.value = '';
      }
    }

    function exportDataBackup() {
      const fullBackup = { app: "BharatWellnessClub", backupDate: new Date().toISOString(), consumers, coaches, attendance: attendanceData, btg: btgData, checkup: checkupData, bmiRecords, salesRecords };
      const downloadAnchor = document.createElement('a');
      downloadAnchor.setAttribute("href", "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(fullBackup, null, 2)));
      downloadAnchor.setAttribute("download", `BWC_Backup_${todayStr}.json`);
      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      downloadAnchor.remove();
    }

    function importDataBackup() {
      const fileInput = document.getElementById('restoreFileInput');
      if (!fileInput.files || fileInput.files.length === 0) return alert('Backup file select karein!');
      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const parsed = JSON.parse(e.target.result);
          if (parsed.consumers) db.ref('consumers').set(parsed.consumers);
          if (parsed.attendance) db.ref('attendance').set(parsed.attendance);
          if (parsed.btg) db.ref('btg').set(parsed.btg);
          if (parsed.checkup) db.ref('checkups').set(parsed.checkup);
          if (parsed.bmiRecords) db.ref('bmiRecords').set(parsed.bmiRecords);
          if (parsed.salesRecords) db.ref('salesRecords').set(parsed.salesRecords);
          if (parsed.coaches) db.ref('coaches').set(parsed.coaches);
          alert('Data restored successfully!');
        } catch(err) { alert('Invalid file!'); }
      };
      reader.readAsText(fileInput.files[0]);
    }
